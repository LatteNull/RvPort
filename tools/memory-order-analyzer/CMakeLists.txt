cmake_minimum_required(VERSION 3.28)
project(memory_order_analyzer VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(LLVM 21.1.8 EXACT REQUIRED CONFIG)
find_library(CLANG_CPP_LIBRARY NAMES clang-cpp clang-cpp.so.21.1 libclang-cpp.so.21.1
             PATHS /usr/lib/llvm-21/lib REQUIRED NO_DEFAULT_PATH)
find_package(GTest REQUIRED)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/protocols/extractor-protocol.json" MOA_PROTOCOL_JSON)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/protocols/extractor-protocol.json")
configure_file(src/protocol.h.in generated/protocol.h @ONLY)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/protocols/analysis-protocol.json" MOA_ANALYSIS_PROTOCOL_JSON)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/protocols/analysis-protocol.json")
configure_file(src/analysis_protocol.h.in generated/analysis_protocol.h @ONLY)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/protocols/thread-analysis-protocol.json" MOA_THREAD_PROTOCOL_JSON)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/protocols/thread-analysis-protocol.json")
configure_file(src/thread_protocol.h.in generated/thread_protocol.h @ONLY)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/protocols/publication-analysis-protocol.json" MOA_PUBLICATION_PROTOCOL_JSON)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/protocols/publication-analysis-protocol.json")
configure_file(src/publication_protocol.h.in generated/publication_protocol.h @ONLY)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/protocols/source-evidence-protocol.json" MOA_SOURCE_PROTOCOL_JSON)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/protocols/source-evidence-protocol.json")
configure_file(src/source_protocol.h.in generated/source_protocol.h @ONLY)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/protocols/bounded-object-provenance-protocol.json" MOA_PROVENANCE_PROTOCOL_JSON)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/protocols/bounded-object-provenance-protocol.json")
configure_file(src/provenance_protocol.h.in generated/provenance_protocol.h @ONLY)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/protocols/demand-driven-analysis-protocol.json" MOA_DEMAND_PROTOCOL_JSON)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/protocols/demand-driven-analysis-protocol.json")
configure_file(src/demand_protocol.h.in generated/demand_protocol.h @ONLY)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/protocols/advanced-atomic-analysis-protocol.json" MOA_ADVANCED_ATOMIC_PROTOCOL_JSON)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/protocols/advanced-atomic-analysis-protocol.json")
configure_file(src/advanced_atomic_protocol.h.in generated/advanced_atomic_protocol.h @ONLY)
option(MOA_COVERAGE "Instrument owned code for unit coverage" ON)
if(MOA_COVERAGE)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "Unit coverage requires the locked Clang compiler")
  endif()
  add_compile_options(-fprofile-instr-generate -fcoverage-mapping)
  add_link_options(-fprofile-instr-generate -fcoverage-mapping)
endif()
add_library(moa_facts src/facts.cpp src/semantic.cpp src/effects_metadata.cpp src/effects_worker.cpp src/effects_local.cpp src/effects_evaluation.cpp src/effect_pointer.cpp src/effect_scalar.cpp src/effect_control.cpp src/effect_interpreter.cpp src/thread_analysis.cpp src/thread_worker.cpp src/publication_analysis.cpp src/publication_worker.cpp src/provenance.cpp src/demand_analysis.cpp)
target_include_directories(moa_facts PUBLIC src "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_include_directories(moa_facts SYSTEM PUBLIC ${LLVM_INCLUDE_DIRS})
target_compile_definitions(moa_facts PUBLIC ${LLVM_DEFINITIONS})
target_link_libraries(moa_facts PUBLIC LLVM)
target_compile_options(moa_facts PRIVATE -Wall -Wextra -Wpedantic)
add_executable(moa-worker src/worker.cpp)
target_link_libraries(moa-worker PRIVATE moa_facts)
add_library(moa_source src/source_facts.cpp)
target_include_directories(moa_source PUBLIC src "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_include_directories(moa_source SYSTEM PUBLIC ${LLVM_INCLUDE_DIRS})
target_compile_definitions(moa_source PUBLIC ${LLVM_DEFINITIONS})
target_link_libraries(moa_source PUBLIC ${CLANG_CPP_LIBRARY} LLVM)
target_compile_options(moa_source PRIVATE -Wall -Wextra -Wpedantic)
add_executable(memory-order-source-worker src/source_worker_main.cpp)
target_link_libraries(memory-order-source-worker PRIVATE moa_source)
enable_testing()
add_executable(moa-unit tests/cpp/test_facts.cpp tests/cpp/test_semantic.cpp tests/cpp/test_p1_compatibility.cpp tests/cpp/test_effects.cpp tests/cpp/test_effect_domain.cpp tests/cpp/test_effect_control.cpp tests/cpp/test_threads.cpp tests/cpp/test_publication.cpp tests/cpp/test_source_facts.cpp tests/cpp/test_provenance.cpp tests/cpp/test_demand_analysis.cpp src/worker_entry.cpp)
target_link_libraries(moa-unit PRIVATE moa_facts moa_source GTest::gtest_main)
target_sources(moa-worker PRIVATE src/worker_entry.cpp)
add_test(NAME unit COMMAND moa-unit)
